==================
Direct and explicit case, catch, and function bodies
==================

fn identity(value)
  value

fn classify(value)
  case value
    of 1
      "one"
    of _ do
      let fallback = "other"
      fallback
    end
  end

fn recover(value)
  do
    raise value
  catch
    of {error = message}
      message
    of _ do
      let fallback = "unknown"
      fallback
    end
  end
