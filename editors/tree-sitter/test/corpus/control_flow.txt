==================
Direct and explicit case, catch, and function bodies
==================

fn identity(value)
  value

fn classify(value)
  case value of
    1 =>
      "one"
    _ =>
      do
        let fallback = "other"
        fallback
      end
  end

fn recover()
  do
    raise "failure"
  catch of
    "failure" =>
      "recovered"
    _ =>
      do
        "fallback"
      end
  end
